# Поддержка форматов библиотеки сверх базовых PDF / DOCX / текста.
#
# Ставится отдельно от requirements.txt: ядро и веб-интерфейс работают без
# этих пакетов, просто читают меньше форматов. Что доступно прямо сейчас,
# показывает команда:  reportgen formats
#
# Помимо пакетов Python нужны внешние программы (их ставят отдельно):
#   LibreOffice  — .doc, .xls, .ppt, .rtf, OpenDocument
#   Tesseract    — распознавание сканов (обязательно с русским языком)
#   DjVuLibre    — .djvu, в нём лежит бо́льшая часть отсканированных книг
#   7-Zip        — архивы .7z и .rar
# Для офлайн-установки они кладутся в комплект: см. docs/15-offline.md.

python-pptx>=0.6.23     # презентации .pptx: слайды, таблицы, заметки докладчика
openpyxl>=3.1           # книги .xlsx/.xlsm: листы, вычисленные значения
xlrd>=2.0               # старые книги .xls (только они, xlsx им не читается)
striprtf>=0.0.26        # .rtf без запуска LibreOffice — быстрее и проще
pillow>=10.0            # подготовка изображений к распознаванию
